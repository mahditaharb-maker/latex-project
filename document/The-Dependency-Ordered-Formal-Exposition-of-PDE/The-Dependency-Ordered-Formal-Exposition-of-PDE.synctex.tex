\bfseries\setstretch{1.5}\begin{abstract}
	\bfseries\setstretch{1.5}\large This document is a step-indexed, dependency-ordered presentation of
	PDE using \cite{evans2010}. The entries are arranged in ten levels of a topological ordering: every result appears only after all results it depends on, with circular dependencies broken at the point where the simpler formulation is introduced first. Each entry is annotated with the choice strength of its proof: \textsf{ZF}, \textsf{ZF+DC},
	or \textsf{ZFC}.
\end{abstract}

\section*{\textsf{Background --- Unproved Calculus Statements}}

\noindent The following block gathers, in one independent list, every calculus
or topology result that is taken as \emph{given} rather than proved in this
document. They are stated here as numbered background assumptions
$B_{1},\dots,B_{5}$, before Level~0, so that the ten levels below depend only
on these assumptions and on the primitives of Level~0. No proof of
$B_{1},\dots,B_{5}$ is given in this document; their choice strength is the
proof-theoretic strength of the standard proofs. Note that
$B_{1},B_{2},B_{3},B_{4}$ are $\mathsf{ZF}$-level facts, whereas $B_{5}$ is a
$\mathsf{ZFC}$-level topology fact (Tychonoff is equivalent to the Axiom of
Choice).

\medskip

\noindent$%
\begin{array}{|r|l|l|l|}
\hline\hline
\refstepcounter{swpentry} \hypertarget{lem:MVT}{\theswpentry} \label%
{lemi:MVT} & B_{1}\;(\mathsf{MVT}):\forall f\in C([a,b])\cap D((a,b)),\;%
\exists c\in(a,b),\;f^{\prime }(c)=(f(b)-f(a))/(b-a) & \mathsf{Mean Value
Theorem}\ \text{\cite{simpson2009}} & \mathsf{ZF} \\ \hline\hline
\refstepcounter{swpentry} \hypertarget{lem:MVTI}{\theswpentry} \label%
{lemi:MVTI} & B_{2}\;(\mathsf{MVTI}):\forall w\ge0,\;g\in C([a,b]),\;%
\exists c\in[a,b],\;\int_{a}^{b}w(t)g(t)\,dt=g(c)\int_{a}^{b}w(t)\,dt & 
\mathsf{Weighted Mean Value Theorem for Integrals} & \mathsf{ZF} \\ 
\hline\hline
\refstepcounter{swpentry} \hypertarget{lem:IBP}{\theswpentry} \label%
{lemi:IBP} & B_{3}\;(\mathsf{IBP}):\int_{a}^{b}u^{\prime }(t)v(t)\,dt=[uv]_{a}%
^{b}-\int_{a}^{b}u(t)v^{\prime }(t)\,dt & \mathsf{Integration by Parts} & 
\mathsf{ZF} \\ \hline\hline
\refstepcounter{swpentry} \hypertarget{lem:BolzanoWeierstrass}{\theswpentry} \label%
{lemi:BolzanoWeierstrass} & B_{4}\;(\mathsf{BW}):\forall \{x_{k}\}\subset%
\mathbb{R}^{n},\;\mathsf{Bdd}(x_{k})\Rightarrow\exists\{x_{k_{j}}\},\;x\in%
\mathbb{R}^{n},\;x_{k_{j}}\to x & \mathsf{Bolzano}-\mathsf{Weierstrass\ Theorem} & 
\mathsf{ZF} \\ \hline\hline
\refstepcounter{swpentry} \hypertarget{lem:Tychonoff}{\theswpentry} \label%
{lemi:Tychonoff} & B_{5}\;(\mathsf{Tych}):\forall \{X_{i}\}_{i\in I},\;%
\bigl(\forall i,\;\mathsf{Compact}(X_{i})\bigr)\Rightarrow\mathsf{Compact}%
\bigl(\textstyle\prod_{i\in I}X_{i}\bigr) & \mathsf{Tychonoff^{\prime }s\ Theorem} & 
\mathsf{ZFC} \\ \hline\hline
\end{array}%
$

\medskip

\noindent\textit{Convention.} All references of the form
$B_{k}$ in what follows point to the block above. No entry of
Levels~0--9 may depend on anything outside
$\{B_{1},B_{2},B_{3},B_{4},B_{5}\}$ plus the
Level~0 primitives. The calculus results
$\mathsf{FTC}$, $\mathsf{Taylor}$, $\mathsf{ProductRule}$, $\mathsf{ChainRule}$,
$\mathsf{JacobiFormula}$, and $\mathsf{JacobiFormulaAdj}$ are \emph{not}
background assumptions; they are proved inside Level~1 below.
